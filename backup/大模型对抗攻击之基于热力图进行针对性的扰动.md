def get_models(cfg: MainConfig):
    """Get models based on configuration.

    Args:
        cfg: Configuration object containing model settings

    Returns:
        Tuple of (feature_extractor, list of models)

    Raises:
        ValueError: If ensemble=False but multiple backbones specified

def to_tensor(pic):

def generate_batch_masks_on_the_fly(path_list, cfg, device):
    masks = []
    for p in path_list:
        raw_image = Image.open(p).convert("RGB")
        mask_image = remove(raw_image, only_mask=True)
        blurred_mask = mask_image.filter(ImageFilter.GaussianBlur(radius=10))
        mask_tensor = transforms.ToTensor()(blurred_mask.convert("L")).to(device)
        mask_tensor = transforms.Resize((cfg.model.input_res, cfg.model.input_res))(mask_tensor)
        masks.append(mask_tensor)
    return torch.stack(masks)

        target_data, batch_size=cfg.data.batch_size, shuffle=False
    )

            source_crop=source_crop,
            img_index=i,

            image_tgt=image_tgt,

            mask=mask  # [新增] 传入 mask

def attack_imgpair(
    cfg: MainConfig,
    ensemble_extractor: nn.Module,
    ensemble_loss: nn.Module,
    source_crop: Optional[transforms.RandomResizedCrop],
    target_crop: Optional[transforms.RandomResizedCrop],
    img_index: int,
    image_org: torch.Tensor,
    path_org: List[str],
    image_tgt: torch.Tensor,
    mask: torch.Tensor  # [新增] 接收 mask
):
    image_org, image_tgt = image_org.to(cfg.model.device), image_tgt.to(
        cfg.model.device
    )
    attack_type = cfg.attack
    attack_fn = {
        "fgsm": fgsm_attack,
        "mifgsm": mifgsm_attack,
        "pgd": pgd_attack,
    }[attack_type]
    adv_image = attack_fn(
        cfg=cfg,
        ensemble_extractor=ensemble_extractor,
        ensemble_loss=ensemble_loss,
        source_crop=source_crop,
        target_crop=target_crop,
        img_index=img_index,
        image_org=image_org,
        image_tgt=image_tgt,
        mask=mask  # [新增] 传入 mask 给具体攻击函数
    )

    # Get config hash for output directory
    config_hash = hash_training_config(cfg)
    unique_hash = f"{config_hash}_{time.strftime('%m%d_%H%M%S')}"

    # Save images
    for path_idx in range(len(path_org)):
        folder, name = (
            path_org[path_idx].split("/")[-2],
            path_org[path_idx].split("/")[-1],
        )
        # Use config hash in output path
        folder_to_save = os.path.join(cfg.data.output, "img", unique_hash, folder)
        ensure_dir(folder_to_save)

        if "JPEG" in name:
            torchvision.utils.save_image(
                adv_image[path_idx], os.path.join(folder_to_save, name[:-4]) + "png"
            )
        elif "png" in name:
            torchvision.utils.save_image(
                adv_image[path_idx], os.path.join(folder_to_save, name)
            )


def log_metrics(pbar, metrics, img_index, epoch=None):
    """
    Log metrics to progress bar and wandb.

    Args:
        pbar: tqdm progress bar to update
        metrics: Dictionary of metrics to log
        img_index: Index of the image (for wandb logging)
        epoch: Optional epoch number for logging
    """
    # Format metrics for progress bar
    pbar_metrics = {
        k: f"{v:.5f}" if "sim" in k else f"{v:.3f}" for k, v in metrics.items()
    }
    pbar.set_postfix(pbar_metrics)

    # Prepare wandb metrics with image index
    wandb_metrics = {f"img{img_index}_{k}": v for k, v in metrics.items()}
    if epoch is not None:
        wandb_metrics["epoch"] = epoch

    # Log to wandb
    wandb.log(wandb_metrics)


def fgsm_attack(
    cfg: MainConfig,
    ensemble_extractor: nn.Module,
    ensemble_loss: nn.Module,
    source_crop: Optional[transforms.RandomResizedCrop],
    target_crop: Optional[transforms.RandomResizedCrop],
    img_index: int,
    image_org: torch.Tensor,
    image_tgt: torch.Tensor,
    mask: torch.Tensor  # [新增] 接收 mask
):
    """
       Perform FGSM attack on the image to generate adversarial examples.

       Args:
           cfg: Configuration parameters
           ensemble_extractor: Ensemble feature extractor model
           ensemble_loss: Ensemble loss function
           source_crop: Optional transform for cropping source images
           target_crop: Optional transform for cropping target images
           i: Index of the image (for logging)
           image_org: Original source image tensor
           image_tgt: Target image tensor to match features with

       Returns:
           torch.Tensor: Generated adversarial image
       """
    # Initialize perturbation
    delta = torch.zeros_like(image_org, requires_grad=True)

    # Progress bar for optimization
    pbar = tqdm(range(cfg.optim.steps), desc=f"Attack progress")

    # Main optimization loop
    for epoch in pbar:

        with torch.no_grad():
            ensemble_loss.set_ground_truth(target_crop(image_tgt))

        # Forward pass
        # [修改点] 将矩阵掩码和生成的扰动相乘，使得只有 mask 区域的特征被抽取用于计算 similarity
        adv_image = image_org + (delta * mask)
        adv_features = ensemble_extractor(adv_image)

        # Calculate metrics (监控实际生效的扰动量)
        metrics = {
            "max_delta": torch.max(torch.abs(delta * mask)).item(),
            "mean_delta": torch.mean(torch.abs(delta * mask)).item(),
        }

        # Calculate loss based on configuration
        global_sim = ensemble_loss(adv_features)
        metrics["global_similarity"] = global_sim.item()

        if cfg.model.use_source_crop:
            # If using source crop, calculate additional local similarity
            local_cropped = source_crop(adv_image)
            local_features = ensemble_extractor(local_cropped)
            local_sim = ensemble_loss(local_features)
            loss = local_sim
            metrics["local_similarity"] = local_sim.item()
        else:
            # Otherwise use global similarity as loss
            loss = global_sim

        # Log current metrics
        log_metrics(pbar, metrics, img_index, epoch)

        grad = torch.autograd.grad(loss, delta, create_graph=False)[0]

        # Update delta using FGSM
        delta.data = torch.clamp(
            delta + cfg.optim.alpha * torch.sign(grad),
            min=-cfg.optim.epsilon,
            max=cfg.optim.epsilon,
        )
        # [新增点] 强制过滤背景梯度，确保 delta 在下一步依然被完全约束
        delta.data = delta.data * mask

    # Create final adversarial image
    # [修改点] 最终输出也加上 mask 约束
    adv_image = image_org + (delta * mask)
    adv_image = torch.clamp(adv_image / 255.0, 0.0, 1.0)

    # Log final perturbation metrics
    final_metrics = {
        "max_delta": torch.max(torch.abs(delta * mask)).item(),
        "mean_delta": torch.mean(torch.abs(delta * mask)).item(),
    }
    log_metrics(pbar, final_metrics, img_index)

    return adv_image


def mifgsm_attack(
    cfg: MainConfig,
    ensemble_extractor: nn.Module,
    ensemble_loss: nn.Module,
    source_crop: Optional[transforms.RandomResizedCrop],
    target_crop: Optional[transforms.RandomResizedCrop],
    img_index: int,
    image_org: torch.Tensor,
    image_tgt: torch.Tensor,
    mask: torch.Tensor  # [新增] 接收 mask
):
    """
        Perform MI-FGSM attack on the image to generate adversarial examples.

        Args:
            cfg: Configuration parameters
            ensemble_extractor: Ensemble feature extractor model
            ensemble_loss: Ensemble loss function
            source_crop: Optional transform for cropping source images
            target_crop: Optional transform for cropping target images
            i: Index of the image (for logging)
            image_org: Original source image tensor
            image_tgt: Target image tensor to match features with

        Returns:
            torch.Tensor: Generated adversarial image
        """
    # Initialize perturbation and momentum
    delta = torch.zeros_like(image_org, requires_grad=True)
    momentum = torch.zeros_like(image_org, requires_grad=False)

    # Progress bar for optimization
    pbar = tqdm(range(cfg.optim.steps), desc=f"Attack progress")

    # Main optimization loop
    for epoch in pbar:

        with torch.no_grad():
            ensemble_loss.set_ground_truth(target_crop(image_tgt))

        # Forward pass
        # [修改点] 将矩阵掩码和生成的扰动相乘
        adv_image = image_org + (delta * mask)
        adv_features = ensemble_extractor(adv_image)

        # Calculate metrics
        metrics = {
            "max_delta": torch.max(torch.abs(delta * mask)).item(),
            "mean_delta": torch.mean(torch.abs(delta * mask)).item(),
        }

        # Calculate loss based on configuration
        global_sim = ensemble_loss(adv_features)
        metrics["global_similarity"] = global_sim.item()

        if cfg.model.use_source_crop:
            # If using source crop, calculate additional local similarity
            local_cropped = source_crop(adv_image)
            local_features = ensemble_extractor(local_cropped)
            local_sim = ensemble_loss(local_features)
            loss = local_sim
            metrics["local_similarity"] = local_sim.item()
        else:
            # Otherwise use global similarity as loss
            loss = global_sim

        log_metrics(pbar, metrics, img_index, epoch)

        grad = torch.autograd.grad(loss, delta, create_graph=False)[0]

        # MI-FGSM update
        momentum = momentum * 0.9 + grad
        delta.data = torch.clamp(
            delta + cfg.optim.alpha * torch.sign(momentum),
            min=-cfg.optim.epsilon,
            max=cfg.optim.epsilon,
        )
        # [新增点] 强制约束 delta
        delta.data = delta.data * mask

    # Create final adversarial image
    # [修改点] 最终输出约束
    adv_image = image_org + (delta * mask)
    adv_image = torch.clamp(adv_image / 255.0, 0.0, 1.0)

    # Log final perturbation metrics
    final_metrics = {
        "max_delta": torch.max(torch.abs(delta * mask)).item(),
        "mean_delta": torch.mean(torch.abs(delta * mask)).item(),
    }
    log_metrics(pbar, final_metrics, img_index)

    return adv_image


def pgd_attack(
    cfg: MainConfig,
    ensemble_extractor: nn.Module,
    ensemble_loss: nn.Module,
    source_crop: Optional[transforms.RandomResizedCrop],
    target_crop: Optional[transforms.RandomResizedCrop],
    img_index: int,
    image_org: torch.Tensor,
    image_tgt: torch.Tensor,
    mask: torch.Tensor  # [新增] 接收 mask
):
    """
       Perform PGD attack on the image to generate adversarial examples.

       Args:
           cfg: Configuration parameters
           ensemble_extractor: Ensemble feature extractor model
           ensemble_loss: Ensemble loss function
           source_crop: Optional transform for cropping source images
           target_crop: Optional transform for cropping target images
           i: Index of the image (for logging)
           image_org: Original source image tensor
           image_tgt: Target image tensor to match features with

       Returns:
           torch.Tensor: Generated adversarial image
       """
    # Initialize perturbation and momentum
    delta = torch.zeros_like(image_org, requires_grad=True)
    optimizer = torch.optim.Adam([delta], lr=cfg.optim.alpha)

    # Progress bar for optimization
    pbar = tqdm(range(cfg.optim.steps), desc=f"Attack progress")

    # Main optimization loop
    for epoch in pbar:

        with torch.no_grad():
            ensemble_loss.set_ground_truth(target_crop(image_tgt))

        # Forward pass
        # [修改点] 将矩阵掩码和生成的扰动相乘
        adv_image = image_org + (delta * mask)
        adv_features = ensemble_extractor(adv_image)

        # Calculate metrics
        metrics = {
            "max_delta": torch.max(torch.abs(delta * mask)).item(),
            "mean_delta": torch.mean(torch.abs(delta * mask)).item(),
        }

        # Calculate loss based on configuration
        global_sim = ensemble_loss(adv_features)
        metrics["global_similarity"] = global_sim.item()

        if cfg.model.use_source_crop:
            # If using source crop, calculate additional local similarity
            local_cropped = source_crop(adv_image)
            local_features = ensemble_extractor(local_cropped)
            local_sim = ensemble_loss(local_features)
            loss = -local_sim # since we want to maximize the loss
            metrics["local_similarity"] = local_sim.item()
        else:
            # Otherwise use global similarity as loss
            loss = -global_sim

        log_metrics(pbar, metrics, img_index, epoch)

        optimizer.zero_grad()
        loss.backward()

        # PGD update
        optimizer.step()
        delta.data = torch.clamp(
            delta,
            min=-cfg.optim.epsilon,
            max=cfg.optim.epsilon,
        )
        # [新增点] 强制约束 delta
        delta.data = delta.data * mask

    # Create final adversarial image
    # [修改点] 最终输出约束
    adv_image = image_org + (delta * mask)
    adv_image = torch.clamp(adv_image / 255.0, 0.0, 1.0)

    # Log final perturbation metrics
    final_metrics = {
        "max_delta": torch.max(torch.abs(delta * mask)).item(),
        "mean_delta": torch.mean(torch.abs(delta * mask)).item(),
    }
    log_metrics(pbar, final_metrics, img_index)

    return adv_image


if __name__ == "__main__":
    main()

def fgsm_attack(
        cfg: MainConfig,
        ensemble_extractor: nn.Module,
        ensemble_loss: nn.Module,
        source_crop: Optional[transforms.RandomResizedCrop],
        target_crop: Optional[transforms.RandomResizedCrop],
        img_index: int,
        image_org: torch.Tensor,
        image_tgt: torch.Tensor,
        weight_mask: torch.Tensor
):
    """
       Perform FGSM attack on the image to generate adversarial examples.

       Args:
           cfg: Configuration parameters
           ensemble_extractor: Ensemble feature extractor model
           ensemble_loss: Ensemble loss function
           source_crop: Optional transform for cropping source images
           target_crop: Optional transform for cropping target images
           i: Index of the image (for logging)
           image_org: Original source image tensor
           image_tgt: Target image tensor to match features with

       Returns:
           torch.Tensor: Generated adversarial image
       """
    # Initialize perturbation
    delta = torch.zeros_like(image_org, requires_grad=True)

    # Progress bar for optimization
    pbar = tqdm(range(cfg.optim.steps), desc=f"Attack progress")

    # Main optimization loop
    for epoch in pbar:

        with torch.no_grad():
            ensemble_loss.set_ground_truth(target_crop(image_tgt))

        # 前向传播：基础扰动 delta 乘以 weight_mask
        adv_image = image_org + (delta * weight_mask)
        adv_features = ensemble_extractor(adv_image)

        actual_delta = delta * weight_mask
        metrics = {
            "max_delta": torch.max(torch.abs(actual_delta)).item(),
            "mean_delta": torch.mean(torch.abs(actual_delta)).item(),
        }

        # Calculate loss based on configuration
        global_sim = ensemble_loss(adv_features)
        metrics["global_similarity"] = global_sim.item()

        if cfg.model.use_source_crop:
            # If using source crop, calculate additional local similarity
            local_cropped = source_crop(adv_image)
            local_features = ensemble_extractor(local_cropped)
            local_sim = ensemble_loss(local_features)
            loss = local_sim
            metrics["local_similarity"] = local_sim.item()
        else:
            # Otherwise use global similarity as loss
            loss = global_sim

        # Log current metrics
        log_metrics(pbar, metrics, img_index, epoch)

        grad = torch.autograd.grad(loss, delta, create_graph=False)[0]

        # 更新基础扰动 delta
        delta.data = torch.clamp(
            delta + cfg.optim.alpha * torch.sign(grad),
            min=-cfg.optim.epsilon,
            max=cfg.optim.epsilon,
        )
        # 【修改二：删除了 delta.data = delta.data * mask】

    # 最终输出依然要乘权重
    adv_image = image_org + (delta * weight_mask)
    adv_image = torch.clamp(adv_image / 255.0, 0.0, 1.0)

    # Log final perturbation metrics
    final_metrics = {
        "max_delta": torch.max(torch.abs(delta * weight_mask)).item(),
        "mean_delta": torch.mean(torch.abs(delta * weight_mask)).item(),
    }
    log_metrics(pbar, final_metrics, img_index)

    return adv_image


def mifgsm_attack(
        cfg: MainConfig,
        ensemble_extractor: nn.Module,
        ensemble_loss: nn.Module,
        source_crop: Optional[transforms.RandomResizedCrop],
        target_crop: Optional[transforms.RandomResizedCrop],
        img_index: int,
        image_org: torch.Tensor,
        image_tgt: torch.Tensor,
        weight_mask: torch.Tensor
):
    """
        Perform MI-FGSM attack on the image to generate adversarial examples.

        Args:
            cfg: Configuration parameters
            ensemble_extractor: Ensemble feature extractor model
            ensemble_loss: Ensemble loss function
            source_crop: Optional transform for cropping source images
            target_crop: Optional transform for cropping target images
            i: Index of the image (for logging)
            image_org: Original source image tensor
            image_tgt: Target image tensor to match features with

        Returns:
            torch.Tensor: Generated adversarial image
        """
    # Initialize perturbation and momentum
    delta = torch.zeros_like(image_org, requires_grad=True)
    momentum = torch.zeros_like(image_org, requires_grad=False)

    # Progress bar for optimization
    pbar = tqdm(range(cfg.optim.steps), desc=f"Attack progress")

    # Main optimization loop
    for epoch in pbar:

        with torch.no_grad():
            ensemble_loss.set_ground_truth(target_crop(image_tgt))

        # 前向传播：基础扰动 delta 乘以 weight_mask
        adv_image = image_org + (delta * weight_mask)
        adv_features = ensemble_extractor(adv_image)

        actual_delta = delta * weight_mask
        metrics = {
            "max_delta": torch.max(torch.abs(actual_delta)).item(),

            local_features = ensemble_extractor(local_cropped)
            local_sim = ensemble_loss(local_features)
            loss = local_sim
            metrics["local_similarity"] = local_sim.item()
        else:
            # Otherwise use global similarity as loss
            loss = global_sim

        log_metrics(pbar, metrics, img_index, epoch)

        grad = torch.autograd.grad(loss, delta, create_graph=False)[0]

        # MI-FGSM update
        momentum = momentum * 0.9 + grad
        delta.data = torch.clamp(
            delta + cfg.optim.alpha * torch.sign(momentum),
            min=-cfg.optim.epsilon,
            max=cfg.optim.epsilon,
        )
        # 【修改二：删除了 delta.data = delta.data * mask】

    adv_image = image_org + (delta * weight_mask)
    adv_image = torch.clamp(adv_image / 255.0, 0.0, 1.0)

    # Log final perturbation metrics
    final_metrics = {
        "max_delta": torch.max(torch.abs(delta * weight_mask)).item(),
        "mean_delta": torch.mean(torch.abs(delta * weight_mask)).item(),
    }
    log_metrics(pbar, final_metrics, img_index)

    return adv_image


def pgd_attack(
        cfg: MainConfig,
        ensemble_extractor: nn.Module,
        ensemble_loss: nn.Module,
        source_crop: Optional[transforms.RandomResizedCrop],
        target_crop: Optional[transforms.RandomResizedCrop],
        img_index: int,
        image_org: torch.Tensor,
        image_tgt: torch.Tensor,
        weight_mask: torch.Tensor
):
    """
       Perform PGD attack on the image to generate adversarial examples.

       Args:
           cfg: Configuration parameters
           ensemble_extractor: Ensemble feature extractor model
           ensemble_loss: Ensemble loss function
           source_crop: Optional transform for cropping source images
           target_crop: Optional transform for cropping target images
           i: Index of the image (for logging)
           image_org: Original source image tensor
           image_tgt: Target image tensor to match features with

       Returns:
           torch.Tensor: Generated adversarial image
       """
    # Initialize perturbation and momentum
    delta = torch.zeros_like(image_org, requires_grad=True)
    optimizer = torch.optim.Adam([delta], lr=cfg.optim.alpha)

    # Progress bar for optimization
    pbar = tqdm(range(cfg.optim.steps), desc=f"Attack progress")

    # Main optimization loop
    for epoch in pbar:

        with torch.no_grad():
            ensemble_loss.set_ground_truth(target_crop(image_tgt))

        # 前向传播：基础扰动 delta 乘以 weight_mask
        adv_image = image_org + (delta * weight_mask)
        adv_features = ensemble_extractor(adv_image)

        actual_delta = delta * weight_mask
        metrics = {
            "max_delta": torch.max(torch.abs(actual_delta)).item(),
            "mean_delta": torch.mean(torch.abs(actual_delta)).item(),
        }

        # Calculate loss based on configuration
        global_sim = ensemble_loss(adv_features)
        metrics["global_similarity"] = global_sim.item()

        if cfg.model.use_source_crop:
            # If using source crop, calculate additional local similarity
            local_cropped = source_crop(adv_image)
            local_features = ensemble_extractor(local_cropped)
            local_sim = ensemble_loss(local_features)
            loss = -local_sim # since we want to maximize the loss
            metrics["local_similarity"] = local_sim.item()
        else:
            # Otherwise use global similarity as loss
            loss = -global_sim

        log_metrics(pbar, metrics, img_index, epoch)

        optimizer.zero_grad()
        loss.backward()

        # PGD update
        optimizer.step()
        delta.data = torch.clamp(
            delta,
            min=-cfg.optim.epsilon,
            max=cfg.optim.epsilon,
        )

    adv_image = image_org + (delta * weight_mask)
    adv_image = torch.clamp(adv_image / 255.0, 0.0, 1.0)

    # Log final perturbation metrics
    final_metrics = {
        "max_delta": torch.max(torch.abs(delta * weight_mask)).item(),
        "mean_delta": torch.mean(torch.abs(delta * weight_mask)).item(),
    }
    log_metrics(pbar, final_metrics, img_index)

    return adv_image

    mode_to_nptype = {"I": np.int32, "I;16": np.int16, "F": np.float32}

def pgd_attack(
        cfg: MainConfig,
        ensemble_extractor: nn.Module,
        ensemble_loss: nn.Module,
        source_crop: Optional[transforms.RandomResizedCrop],
        target_crop: Optional[transforms.RandomResizedCrop],
        img_index: int,
        image_org: torch.Tensor,
        image_tgt: torch.Tensor,
        mask: torch.Tensor
):
    delta = torch.zeros_like(image_org, requires_grad=True)
    optimizer = torch.optim.Adam([delta], lr=cfg.optim.alpha)
    pbar = tqdm(range(cfg.optim.steps), desc=f"Attack progress")

    for epoch in pbar:
        with torch.no_grad():
            ensemble_loss.set_ground_truth(target_crop(image_tgt))

        # ================= 第一阶段：全局扰动 =================
        adv_image_1 = image_org + delta
        adv_features_1 = ensemble_extractor(adv_image_1)
        global_sim_1 = ensemble_loss(adv_features_1)

        if cfg.model.use_source_crop:
            local_cropped_1 = source_crop(adv_image_1)
            local_features_1 = ensemble_extractor(local_cropped_1)
            loss_1 = -ensemble_loss(local_features_1)  # PGD requires negative loss to maximize similarity
        else:
            loss_1 = -global_sim_1

        optimizer.zero_grad()
        loss_1.backward()
        # 第一阶段：全局 Adam 步进
        optimizer.step()
        delta.data = torch.clamp(delta, min=-cfg.optim.epsilon, max=cfg.optim.epsilon)

        # ================= 第二阶段：局部重点扰动 =================
        adv_image_2 = image_org + delta
        adv_features_2 = ensemble_extractor(adv_image_2)
        global_sim_2 = ensemble_loss(adv_features_2)

        if cfg.model.use_source_crop:
            local_cropped_2 = source_crop(adv_image_2)
            local_features_2 = ensemble_extractor(local_cropped_2)
            loss_2 = -ensemble_loss(local_features_2)
            local_sim_val = -loss_2.item()
        else:
            loss_2 = -global_sim_2
            local_sim_val = None

        optimizer.zero_grad()
        loss_2.backward()

        # 核心逻辑：拦截 Adam 的全图更新，只让 mask 区域生效
        delta_before = delta.data.clone()
        optimizer.step()
        delta_update = delta.data - delta_before

        # 第二次更新仅附加在重点区域
        delta.data = delta_before + (delta_update * mask)
        delta.data = torch.clamp(delta, min=-cfg.optim.epsilon, max=cfg.optim.epsilon)

        # ================= 记录指标 =================
        metrics = {
            "max_delta": torch.max(torch.abs(delta)).item(),
            "mean_delta": torch.mean(torch.abs(delta)).item(),
            "global_similarity": global_sim_2.item()
        }
        if local_sim_val is not None:
            metrics["local_similarity"] = local_sim_val

        log_metrics(pbar, metrics, img_index, epoch)

    adv_image = image_org + delta
    adv_image = torch.clamp(adv_image / 255.0, 0.0, 1.0)
    return adv_image
