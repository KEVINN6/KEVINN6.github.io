大模型对抗攻击之热力图的生成

1.1 初步实现方案

1. 热力图是怎么来的？

在 Vision Transformer (ViT) 架构中，图像被切分成了一个个 Patch（比如 $16 \times 16$ 像素）。模型在处理这些 Patch 时，会额外加入一个特殊的 [CLS] Token。这个 [CLS] Token 的最终输出通常被用来表示整张图像的全局特征。在 Transformer 的最后一层，[CLS] Token 会通过注意力机制（Attention）与所有的图像 Patch 进行信息交互。[CLS] Token 对各个 Patch 的注意力权重大小，实际上就代表了该 Patch 对整张图像语义的贡献程度。 将这些权重提取出来并重新拼接成一个二维网格，就是一张完美的语义热力图（Attention Map）。

2. 如何提取？

直接使用 Hugging Face 的 transformers 库，加载开源的 OpenAI CLIP 模型来实现。这与论文中使用的替代模型是一致的。

import os
import random
import numpy as np
import torch
import torchvision
from PIL import Image
import hydra
from omegaconf import DictConfig
import matplotlib.pyplot as plt
import cv2

# 引入 Hugging Face 的库用于提取 Attention
from transformers import CLIPVisionModel, CLIPProcessor

from config_schema import MainConfig
from utils import setup_wandb, ensure_dir


# 保留原有的数据集加载逻辑
class ImageFolderWithPaths(torchvision.datasets.ImageFolder):
    def __getitem__(self, index):
        original_tuple = super().__getitem__(index)
        path, _ = self.samples[index]
        return original_tuple + (path,)


def set_environment(seed=2023):
    random.seed(seed)
    os.environ["PYTHONHASHSEED"] = str(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    torch.cuda.manual_seed(seed)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False


def generate_and_save_heatmap(image_path, model, processor, output_path, device):
    """
    核心：利用 CLIP 模型提取目标图像的注意力热力图并保存
    """
    # 1. 读取原始图像并进行 CLIP 标准预处理
    raw_image = Image.open(image_path).convert("RGB")
    inputs = processor(images=raw_image, return_tensors="pt").to(device)

    # 2. 前向传播，强制要求模型返回 attention 权重
    with torch.no_grad():
        outputs = model(**inputs, output_attentions=True)

    # 3. 提取最后一层的注意力矩阵
    # 形状为: (batch_size, num_heads, seq_len, seq_len)
    last_layer_attention = outputs.attentions[-1]

    # 4. 获取 [CLS] token (索引为0) 对其他所有 patch (索引 1 开始) 的注意力，并在所有的 Head 上求平均
    # 结果是一个长度为 196 的一维张量 (对于 ViT-B/16)
    cls_attention = last_layer_attention[0, :, 0, 1:].mean(dim=0)

    # 5. 将一维注意力转换回二维空间网格 (14 x 14)
    grid_size = int(np.sqrt(cls_attention.shape[0]))
    attention_map = cls_attention.reshape(grid_size, grid_size).cpu().numpy()

    # 6. 归一化处理 (0 到 1 之间)
    attention_map = (attention_map - attention_map.min()) / (attention_map.max() - attention_map.min())

    # 7. 将 14x14 的热力图放大回原图的尺寸
    original_size = raw_image.size  # (width, height)
    heatmap_resized = cv2.resize(attention_map, original_size, interpolation=cv2.INTER_LANCZOS4)

    # 8. 可视化：将热力图与原图融合
    plt.figure(figsize=(10, 5))

    # 左图：原图
    plt.subplot(1, 2, 1)
    plt.imshow(raw_image)
    plt.axis('off')
    plt.title("Target Image")

    # 右图：热力图叠加
    plt.subplot(1, 2, 2)
    plt.imshow(raw_image)
    # 使用 jet 伪彩色映射，alpha 控制透明度
    plt.imshow(heatmap_resized, cmap='jet', alpha=0.5)
    plt.axis('off')
    plt.title("Attention Heatmap")

    # 确保输出目录存在并保存
    ensure_dir(os.path.dirname(output_path))
    plt.savefig(output_path, bbox_inches='tight', pad_inches=0.1)
    plt.close()


@hydra.main(version_base=None, config_path="config", config_name="ensemble_3models")
def main(cfg: MainConfig):
    set_environment()
    device = cfg.model.device if hasattr(cfg.model, 'device') else "cuda"

    print("Loading CLIP Model for Heatmap Extraction...")
    # 我们这里显式加载一个标准的 ViT-B/16 用于提取热力图
    # 这与 M-Attack 集成模型中的 B16 是同源的
    model_name = "openai/clip-vit-base-patch16"
    clip_model = CLIPVisionModel.from_pretrained(
        model_name,
        attn_implementation="eager"  # 强制使用传统的注意力机制以提取热力图
    ).to(device).eval()
    clip_processor = CLIPProcessor.from_pretrained(model_name)

    # 复用你的 Target Dataloader，只为了拿到原图的物理路径
    # 因为自定义的 to_tensor 可能改变了像素分布，直接用路径读图最稳妥
    target_data = ImageFolderWithPaths(cfg.data.tgt_data_path, transform=torchvision.transforms.ToTensor())
    data_loader_target = torch.utils.data.DataLoader(target_data, batch_size=1, shuffle=False)

    for i, (image_tgt, _, path_tgt) in enumerate(data_loader_target):
        if i >= cfg.data.num_samples:
            break

        current_path = path_tgt[0]  # batch size 为 1
        folder = current_path.split("/")[-2]
        name = current_path.split("/")[-1].rsplit('.', 1)[0]  # 去掉后缀

        # 定义热力图的保存路径
        output_path = os.path.join(cfg.data.output, "heatmaps", folder, f"{name}_heatmap.png")

        print(f"[{i + 1}/{cfg.data.num_samples}] Generating heatmap for {name}...")

        generate_and_save_heatmap(
            image_path=current_path,
            model=clip_model,
            processor=clip_processor,
            output_path=output_path,
            device=device
        )

    print(f"All heatmaps saved to {os.path.join(cfg.data.output, 'heatmaps')}")


if __name__ == "__main__":
    main()

2.1.1 改进点 ：融合多层注意力 + 空间平滑

不要只取最后一层（outputs.attentions[-1]），而是把后面几层（比如最后 4 层，或者全部 12 层）的注意力取平均。底层的注意力通常更关注物体的边缘和轮廓，高层关注抽象语义，融合后能得到完整的主体覆盖。

def generate_and_save_heatmap(image_path, text_prompt, model, processor, output_path, device):
    """
    核心：通过计算 文本特征 与 图像 Patch 特征 的余弦相似度来生成精准的语义热力图
    """
    raw_image = Image.open(image_path).convert("RGB")
    
    # 1. 同时输入文本和图像
    inputs = processor(
        text=[text_prompt], 
        images=raw_image, 
        return_tensors="pt", 
        padding=True
    ).to(device)

    with torch.no_grad():
        # 2. 获取文本特征并进行 L2 归一化
        text_features = model.get_text_features(input_ids=inputs.input_ids, attention_mask=inputs.attention_mask)
        text_features = text_features / text_features.norm(p=2, dim=-1, keepdim=True) # shape: (1, 512)

        # 3. 获取图像的 Patch 特征
        # vision_model 输出的 last_hidden_state 包含了 [CLS] 和所有 Patch
        vision_outputs = model.vision_model(pixel_values=inputs.pixel_values)
        
        # 去掉第 0 个 [CLS] token，保留后面的 196 个 Patch 特征 (对于 ViT-B/16)
        patch_hidden_states = vision_outputs.last_hidden_state[:, 1:, :] # shape: (1, 196, 768)

        # 4. 将 Patch 特征投影到与文本相同的多模态嵌入空间
        patch_embeds = model.visual_projection(patch_hidden_states) # shape: (1, 196, 512)

        # 5. 对 Patch 特征进行 L2 归一化
        patch_embeds = patch_embeds / patch_embeds.norm(p=2, dim=-1, keepdim=True)

        # 6. 计算每个 Patch 与文本的余弦相似度
        # (1, 196, 512) 与 (1, 512, 1) 做矩阵乘法 -> 得到 (1, 196, 1) 的相似度分数
        similarities = torch.bmm(patch_embeds, text_features.unsqueeze(2)).squeeze() # shape: (196,)

        # 7. 将相似度分数重塑为 2D 网格
        grid_size = int(np.sqrt(similarities.shape[0]))
        attention_map = similarities.reshape(grid_size, grid_size).cpu().numpy()

        # 8. 过滤掉负相关区域，并归一化 (关键：排除无关的背景噪声)
        attention_map = np.maximum(attention_map, 0) 
        if attention_map.max() > 0:
            attention_map = attention_map / attention_map.max()

        # 9. 空间平滑
        attention_map = attention_map.astype(np.float32)
        attention_map = cv2.GaussianBlur(attention_map, (3, 3), 0)
        attention_map = np.clip(attention_map, 0, 1)

        # 10. 放大回原图尺寸
        original_size = raw_image.size 
        heatmap_resized = cv2.resize(attention_map, original_size, interpolation=cv2.INTER_LANCZOS4)

    # 11. 可视化绘制
    plt.figure(figsize=(10, 5))
    plt.subplot(1, 2, 1)
    plt.imshow(raw_image)
    plt.axis('off')
    plt.title("Target Image")

    plt.subplot(1, 2, 2)
    plt.imshow(raw_image)
    plt.imshow(heatmap_resized, cmap='jet', alpha=0.5)
    plt.axis('off')
    
    # 标题显示出我们用来引导的关键词
    plt.title(f"Keywords: {text_prompt.replace('a photo of ', '')}", fontsize=10)

    ensure_dir(os.path.dirname(output_path))
    plt.savefig(output_path, bbox_inches='tight', pad_inches=0.1)
    plt.close()

def main(cfg: MainConfig):
    set_environment()
    device = cfg.model.device if hasattr(cfg.model, 'device') else "cuda"

    # ================= 1. 读取 JSON 构建关键词字典 =================
    # 使用 to_absolute_path 防止 hydra 运行时改变工作目录导致找不到文件
    json_path = to_absolute_path("resources/images/target_images/1/keywords.json")
    
    keyword_dict = {}
    if os.path.exists(json_path):
        print(f"Loading keywords from: {json_path}")
        with open(json_path, 'r', encoding='utf-8') as f:
            keywords_data = json.load(f)
            # 建立映射: 去掉后缀的文件名 -> "keyword1, keyword2..."
            # 例如: "1.jpg" -> {"1": "ski, skiers, gear, snow"}
            for item in keywords_data:
                base_name = os.path.splitext(item['image'])[0]
                keyword_dict[base_name] = ", ".join(item['keywords'])
    else:
        print(f"WARNING: Keywords file not found at {json_path}")
    # ================================================================

    print("Loading CLIP Model for Heatmap Extraction...")
    model_name = "openai/clip-vit-base-patch16"
    
    # 这次加载完整的 CLIPModel
    clip_model = CLIPModel.from_pretrained(model_name).to(device).eval()
    clip_processor = CLIPProcessor.from_pretrained(model_name)

    target_data = ImageFolderWithPaths(cfg.data.tgt_data_path, transform=torchvision.transforms.ToTensor())
    data_loader_target = torch.utils.data.DataLoader(target_data, batch_size=1, shuffle=False)

    for i, (image_tgt, _, path_tgt) in enumerate(data_loader_target):
        if i >= cfg.data.num_samples:
            break

        current_path = path_tgt[0] 
        folder = current_path.split("/")[-2]
        filename = current_path.split("/")[-1]
        base_name = os.path.splitext(filename)[0]
        
        output_path = os.path.join(cfg.data.output, "heatmaps_text_guided", folder, f"{base_name}_heatmap.png")
        
        # ================= 2. 动态生成提示词 =================
        if base_name in keyword_dict:
            # 加入 "a photo of" 前缀，符合 CLIP 预训练习惯，效果最好
            text_prompt = f"a photo of {keyword_dict[base_name]}"
        else:
            # Fallback：如果没有找到关键词，使用通用提示词
            text_prompt = "the main subjects, people, and objects in this photo"
        # =======================================================
            
        print(f"[{i+1}/{cfg.data.num_samples}] Generating text-guided heatmap for {filename}...")
        print(f"  -> Prompt: '{text_prompt}'")
        
        generate_and_save_heatmap(
            image_path=current_path,
            text_prompt=text_prompt,
            model=clip_model,
            processor=clip_processor,
            output_path=output_path,
            device=device
        )

    print(f"All heatmaps saved to {os.path.join(cfg.data.output, 'heatmaps_text_guided')}")

def get_next_run_directory(base_path):
    """
    自动获取下一个可用的数字文件夹路径，防止结果覆盖
    """
    ensure_dir(base_path)
    existing_ids = [int(d) for d in os.listdir(base_path) if d.isdigit()]
    next_id = max(existing_ids + [0]) + 1
    new_dir = os.path.join(base_path, str(next_id))
    return new_dir, next_id


def generate_and_save_heatmap(image_path, output_path):
    """
    核心：利用 U2-Net (rembg) 提取主体 Mask，并使用 PIL 生成平滑的热力图
    """
    raw_image = Image.open(image_path).convert("RGB")

    # 1. 使用 rembg 提取主体掩码 (返回的是 PIL Image 对象)
    mask_image = remove(raw_image, only_mask=True)

    # 2. 空间平滑 (使用 PIL 自带的高斯模糊，完全抛弃 cv2)
    # radius=10 大约等同于之前 cv2 的 (21, 21) 模糊核
    blurred_mask = mask_image.filter(ImageFilter.GaussianBlur(radius=10))

    # 3. 将模糊后的 Mask 转为 numpy 数组并归一化为 0~1 的概率矩阵
    attention_map = np.array(blurred_mask).astype(np.float32) / 255.0
    attention_map = np.clip(attention_map, 0, 1)

    # (rembg 返回的 mask 尺寸默认与原图一致，所以连 cv2.resize 也省了！)

    # 4. 可视化绘制并保存
    plt.figure(figsize=(10, 5))

    plt.subplot(1, 2, 1)
    plt.imshow(raw_image)
    plt.axis('off')
    plt.title("Target Image")

    plt.subplot(1, 2, 2)
    plt.imshow(raw_image)
    plt.imshow(attention_map, cmap='jet', alpha=0.5)
    plt.axis('off')
    plt.title("Saliency Object Heatmap (U2-Net)")

    # 确保输出目录存在并保存
    ensure_dir(os.path.dirname(output_path))
    plt.savefig(output_path, bbox_inches='tight', pad_inches=0.1)
    plt.close()


@hydra.main(version_base=None, config_path="config", config_name="ensemble_3models")
def main(cfg: MainConfig):
    set_environment()

    # ================= 1. 确定本次运行的输出目录 =================
    heatmaps_root = to_absolute_path(os.path.join(cfg.data.output, "heatmaps_saliency"))
    current_run_dir, run_id = get_next_run_directory(heatmaps_root)
    print(f"Current run output directory: {current_run_dir} (Run ID: {run_id})")
    # ==========================================================

    print("Initializing U2-Net Model for Saliency Object Extraction...")

    # 复用 Target Dataloader 拿原图路径
    target_data = ImageFolderWithPaths(cfg.data.tgt_data_path, transform=torchvision.transforms.ToTensor())
    data_loader_target = torch.utils.data.DataLoader(target_data, batch_size=1, shuffle=False)

    for i, (image_tgt, _, path_tgt) in enumerate(data_loader_target):
        if i >= cfg.data.num_samples:
            break

        current_path = path_tgt[0]
        folder_name = current_path.split("/")[-2]
        filename = current_path.split("/")[-1]
        base_name = os.path.splitext(filename)[0]

        output_path = os.path.join(current_run_dir, folder_name, f"{base_name}_heatmap.png")

        print(f"[{i + 1}/{cfg.data.num_samples}] Extracting saliency heatmap for {filename} -> Run {run_id}")

        generate_and_save_heatmap(
            image_path=current_path,
            output_path=output_path
        )

    print(f"All results for this run are perfectly saved to: {current_run_dir}")


if __name__ == "__main__":
    main()
